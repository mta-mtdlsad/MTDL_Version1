package com.mtms.repository;

import com.domain.com.mtms.Alert;
import com.mtms.util.DbConnection;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.Timestamp;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class AlertRepository {

    private static Connection connection = DbConnection.getConnection();
    private static PreparedStatement preparedStatement = null;
    private static List<Alert> listOfAlerts = null;


    // Insert an alert into the database
    public static void addAlert(Alert alert) {
        try {
            preparedStatement = connection.prepareStatement("INSERT INTO ALERTS " + "(track_number, alert_message, alert_severity, alert_time, source_id, event_type) " +
                    "VALUES (?, ?, ?, ?, ?, ?)");

            preparedStatement.setString(1, alert.getTrackNumber());
            preparedStatement.setString(2, alert.getAlertMessage());
            preparedStatement.setString(3, alert.getAlertSeverity());
            preparedStatement.setTimestamp(4, Timestamp.valueOf(alert.getAlertTime()));
            preparedStatement.setString(5, alert.getSourceId());
            preparedStatement.setString(6, alert.getEventType());
            preparedStatement.executeUpdate();
            System.out.println("Alert added successfully");
        } catch (Exception e) {
            System.out.println("Alert details not inserted: " + e);
            e.printStackTrace();
        }
    }


    // Insert an alert using an existing database transaction
    public static void addAlert(Connection transactionConnection, Alert alert) throws SQLException {

        PreparedStatement alertStatement = transactionConnection.prepareStatement("INSERT INTO ALERTS " + "(track_number, alert_message, alert_severity, alert_time, source_id, event_type) " +
                "VALUES (?, ?, ?, ?, ?, ?)");
        try {
            alertStatement.setString(1, alert.getTrackNumber());
            alertStatement.setString(2, alert.getAlertMessage());
            alertStatement.setString(3, alert.getAlertSeverity());
            alertStatement.setTimestamp(4, Timestamp.valueOf(alert.getAlertTime()));
            alertStatement.setString(5, alert.getSourceId());
            alertStatement.setString(6, alert.getEventType());
            alertStatement.executeUpdate();
            System.out.println("Alert added successfully");
        } finally {
            alertStatement.close();
        }
    }

    // Retrieve all alerts
    public static List<Alert> getAllAlerts() {
        try {
            preparedStatement = connection.prepareStatement("SELECT * FROM ALERTS");
            ResultSet resultSet = preparedStatement.executeQuery();
            listOfAlerts = new ArrayList<Alert>();
            while (resultSet.next()) {
                listOfAlerts.add(
                        new Alert(
                                resultSet.getInt("alert_id"),
                                resultSet.getString("track_number"),
                                resultSet.getString("alert_message"),
                                resultSet.getString("alert_severity"),
                                resultSet.getTimestamp("alert_time").toLocalDateTime(),
                                resultSet.getString("source_id"),
                                resultSet.getString("event_type")
                        ));
            }

            if (!listOfAlerts.isEmpty())
                return listOfAlerts;
        } catch (Exception exy) {
            System.out.println("Unable to fetch all alerts");
            exy.printStackTrace();
        }
        return null;
    }


    // Select to count alerts by severity
    public Map<String, Integer> getAlertSeverity() {
        Map<String, Integer> severitiesCount = new LinkedHashMap<>();
        try {
            preparedStatement = connection.prepareStatement("SELECT alert_severity, COUNT(*) FROM ALERTS GROUP BY alert_severity");
            ResultSet resultSetAlert = preparedStatement.executeQuery();
            while (resultSetAlert.next()) {
                String alertSeverity = resultSetAlert.getString(1);
                Integer severityCount = resultSetAlert.getInt(2);
                severitiesCount.put(alertSeverity, severityCount);
            }
        } catch (Exception exs) {
            System.out.println("Unable to fetch alert severity details");
            exs.printStackTrace();
        }
        return severitiesCount;
    }
}

package com.mtms.repository;

import com.domain.com.mtms.Source;
import com.domain.com.mtms.Alert;
import com.mtms.util.DbConnection;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.Timestamp;
import java.time.LocalDateTime;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;


public class SourceRepository {

    private static Connection connection = DbConnection.getConnection();
    private static PreparedStatement preparedStatement = null;
    private static List<Source> listOfSources = null;

    // Register a new source
    public static void registerSource(Source source) {
        try {
            preparedStatement = connection.prepareStatement(
                    "INSERT INTO SOURCE " +
                    "(source_id, first_seen_at, last_received_at, latest_sequence, source_status) "
                    + "VALUES (?, ?, ?, ?, ?) ON CONFLICT (source_id) DO NOTHING");
            preparedStatement.setString(1, source.getSourceId());
            preparedStatement.setTimestamp(2, Timestamp.valueOf(source.getFirstSeenAt()));
            // last_received_at by default is null, that's why we use if statement
            if (source.getLastReceivedAt() != null) {
                preparedStatement.setTimestamp(3, Timestamp.valueOf(source.getLastReceivedAt()));
            } else {
                preparedStatement.setTimestamp(3, null);
            }
            preparedStatement.setLong(4, source.getLatestSequence());
            preparedStatement.setString(5, source.getSourceStatus());
            preparedStatement.executeUpdate();
            System.out.println("Source registered successfully");
        } catch (Exception e) {
            System.out.println("Source details not inserted: " + e);
            e.printStackTrace();
        }
    }

    // Retrieve all registered sources
    public static List<Source> getAllSources() {
        try {
            preparedStatement = connection.prepareStatement("SELECT * FROM SOURCE");
            ResultSet resultSet = preparedStatement.executeQuery();
            listOfSources = new ArrayList<Source>();

            while (resultSet.next()) {
                Timestamp lastReceived = resultSet.getTimestamp("last_received_at");
                listOfSources.add(new Source(
                        resultSet.getString("source_id"),
                        resultSet.getTimestamp("first_seen_at").toLocalDateTime(), lastReceived != null ? lastReceived.toLocalDateTime() : null,
                        resultSet.getLong("latest_sequence"),
                        resultSet.getString("source_status")));
            }
            if (!listOfSources.isEmpty())
                return listOfSources;
        } catch (Exception exy) {
            System.out.println("Unable to fetch all sources");
            exy.printStackTrace();
        }
        return null;
    }

    // Retrieve a source using its ID
    public static Source getSourceById(String sourceId) {
        try {
            preparedStatement = connection.prepareStatement("SELECT * FROM SOURCE WHERE source_id = ?");
            preparedStatement.setString(1, sourceId);
            ResultSet resultSet = preparedStatement.executeQuery();
            if (resultSet.next()) {
                Timestamp lastReceived = resultSet.getTimestamp("last_received_at");
                return new Source(
                        resultSet.getString("source_id"),
                        resultSet.getTimestamp("first_seen_at").toLocalDateTime(), lastReceived != null ? lastReceived.toLocalDateTime() : null,
                        resultSet.getLong("latest_sequence"),
                        resultSet.getString("source_status"));
            }
        } catch (Exception e) {
            System.out.println("Unable to retrieve source: " + e);
            e.printStackTrace();
        }
        return null;
    }

    // Update the last received time
    public static void updateLastReceivedTime(String sourceId, LocalDateTime receivedTime) {
        try {
            preparedStatement = connection.prepareStatement("UPDATE SOURCE SET last_received_at = ?, source_status = 'ACTIVE' " + "WHERE source_id = ?");
            preparedStatement.setTimestamp(1, Timestamp.valueOf(receivedTime));
            preparedStatement.setString(2, sourceId);
            preparedStatement.executeUpdate();
            System.out.println("Source receive time updated successfully");
        } catch (Exception e) {
            System.out.println("Unable to update receive time: " + e);
            e.printStackTrace();
        }
    }

    // Update sequence number only if it is greater than the latest sequence
    public static boolean updateLatestSequenceIfValid(String sourceId, long sequenceNumber) {
        try {
            preparedStatement = connection.prepareStatement("UPDATE SOURCE SET latest_sequence = ? " + "WHERE source_id = ? AND latest_sequence < ?");
            preparedStatement.setLong(1, sequenceNumber);
            preparedStatement.setString(2, sourceId);
            preparedStatement.setLong(3, sequenceNumber);
            int rowsUpdated = preparedStatement.executeUpdate();
            return rowsUpdated == 1;
        } catch (Exception e) {
            System.out.println("Unable to update sequence number: " + e);
            e.printStackTrace();
        }
        return false;
    }

    // Calculate source status using the latest receive time
    private static String calculateStatus(LocalDateTime lastReceived, LocalDateTime checkTime) {
        if (lastReceived == null) {
            return "NOT SEEN";
        }
        Duration timeDifference = Duration.between(lastReceived, checkTime);
        if (timeDifference.compareTo(Duration.ofSeconds(15)) <= 0) {
            return "ACTIVE";
        } else if (timeDifference.compareTo(Duration.ofSeconds(60)) <= 0) {
            return "STALE";
        } else {
            return "DISCONNECTED";
        }
    }

    // Update source status and insert its alert in one transaction
    public static boolean updateSourceStatus(String sourceId, String oldStatus, String newStatus, LocalDateTime checkTime, Alert alert) {
        try {
            // Disable auto-commit to start a database transaction
            // This allows the source status update and alert insertion to be saved together
            connection.setAutoCommit(false);
            // Retrieve the last received time and current status of the source
            // FOR UPDATE locks the selected row to prevent simultaneous status changes
            preparedStatement = connection.prepareStatement("SELECT last_received_at, source_status FROM SOURCE WHERE source_id = ? FOR UPDATE");
            preparedStatement.setString(1, sourceId);
            ResultSet resultSet = preparedStatement.executeQuery();
            // Check whether the source exists in the database
            // If it does not exist, close resources, cancel the transaction and return false
            if (!resultSet.next()) {
                resultSet.close();
                preparedStatement.close();
                connection.rollback();
                return false;
            }
            // Retrieve the last received time and current source status
            Timestamp timestamp = resultSet.getTimestamp("last_received_at");
            String currentStatus = resultSet.getString("source_status");
            // Convert the last received time into LocalDateTime
            // If it is NULL, the source has never received a message
            LocalDateTime lastReceived = null;
            if (timestamp != null) {
                lastReceived = timestamp.toLocalDateTime();
            }
            resultSet.close();
            preparedStatement.close();
            // Check whether the current database status matches the old status
            // If another operation has already changed the status, cancel the transaction
            if (currentStatus == null || !currentStatus.equals(oldStatus)) {
                connection.rollback();
                return false;
            }
            // Recalculate the source status using its latest receive time
            // This ensures that the new status is still correct
            String actualStatus = calculateStatus(lastReceived, checkTime);
            // Check whether the calculated status matches the expected new status
            // If they are different, cancel the transaction
            if (!actualStatus.equals(newStatus)) {
                connection.rollback();
                return false;
            }
            // Update the source status in the SOURCE table
            preparedStatement = connection.prepareStatement("UPDATE SOURCE SET source_status = ? WHERE source_id = ?");
            preparedStatement.setString(1, newStatus);
            preparedStatement.setString(2, sourceId);
            preparedStatement.executeUpdate();
            preparedStatement.close();
            // Insert an alert if one was provided
            // The alert uses the same database transaction as the source update
            // alert only when the source changes to STALE or DISCONNECTED
            if (alert != null &&
                    (newStatus.equals("STALE") || newStatus.equals("DISCONNECTED")) &&
                    !oldStatus.equals(newStatus)) {

                AlertRepository.addAlert(connection, alert);
            }
            // Commit the transaction
            // Save the source status update and alert insertion together
            connection.commit();
            System.out.println("Source status updated successfully");
            return true;
        } catch (Exception e) {
            // If an error occurs, cancel all uncommitted database changes
            try {
                connection.rollback();
            } catch (Exception ex) {
                System.out.println(ex);
            }
            System.out.println("Unable to update source status");
            return false;
        }
    }
}

package com.mtms.service;

public class AlertService {

    static AlertRepository alertRepository = new AlertRepository();


    // Process alert and save in repository
    public static void processAlert(String sourceId, String trackNumber, String alertMessage, String eventType) {

        // processes short track number
        if (alertMessage.contains("short"))
            trackNumber = "INVALID(" + trackNumber + ")";

        // create alert object and save to repository
        createAlertObject(
                trackNumber,
                alertMessage,
                chooseSeverity(eventType), // assigning severity base on event type
                sourceId,
                eventType);

    private static String chooseSeverity(String eventType) {
        return switch (eventType) {
            case "Malformed message", "Source becomes disconnected"  -> "HIGH";
            case "Track validation failure", "Source becomes stale" -> "MEDIUM";
            case "Duplicate message", "Out-of-order message" -> "LOW";

        };
    }

        Alert alert = new Alert(
                trackNumber,
                alertMessage,
                severity,
                LocalDateTime.now(),
                sourceId,
                eventType
        );
        AlertRepository.addAlert(alert);
    }

package com.mtms.service;

import com.domain.com.mtms.Source;
import com.domain.com.mtms.Alert;
import com.domain.com.mtms.Track;
import com.repository.com.mtms.SourceRepository;
import com.repository.com.mtms.AlertRepository;
import java.time.LocalDateTime;
import java.time.Duration;
import java.util.List;

public class SourceService {

    // Main method to process incoming tracking data
    public static void processSources(Track track) {
        if (track == null) {
            System.out.println("Track data is null");
            return;
        }

        // Extract source information from Track
        String sourceId = track.getSourceId();
        long sequenceNumber = track.getSequenceNumber();
        String trackNumber = track.getTrackNumber();

        // Process the received message
        boolean received = receiveMessage(sourceId, sequenceNumber);
        if (!received) {
            System.out.println("Unable to process message from: " + sourceId);
        }
        // Check the health of all registered sources
        checkSourceHealth(trackNumber);
    }

    // Register a new source
    public static void registerSource(String sourceId) {
        Source existingSource = SourceRepository.getSourceById(sourceId);
        if (existingSource == null) {
            Source source = new Source(sourceId, LocalDateTime.now(), null, 0, "NOT SEEN");
            SourceRepository.registerSource(source);
            System.out.println("New source registered: " + sourceId);
        } else {
            System.out.println("Source already registered: " + sourceId);
        }
    }

    // Handle a received message from a source
    public static boolean receiveMessage(String sourceId, long sequenceNumber) {
        LocalDateTime receivedTime = LocalDateTime.now();
        Source source = SourceRepository.getSourceById(sourceId);

        // Automatically register new sources
        if (source == null) {
            registerSource(sourceId);
            source = SourceRepository.getSourceById(sourceId);
            if (source == null) {
                System.out.println("Unable to register source");
                return false;
            }
        }

        // Validate sequence number before updating receive time
        if (sequenceNumber <= 0) {
            System.out.println("Invalid sequence number from: " + sourceId);
            return false;
        }

        // Update sequence number atomically
        boolean sequenceUpdated =
                SourceRepository.updateLatestSequenceIfValid(sourceId, sequenceNumber);

        if (!sequenceUpdated) {
            System.out.println("Duplicate or out-of-order message from: " + sourceId);
            return false;
        }

        // Update receive time after successful validation
        SourceRepository.updateLastReceivedTime(sourceId, receivedTime);
        System.out.println("Message received successfully from: " + sourceId);
        return true;
    }

    // Calculate source health status
    private static String calculateSourceStatus(Source source, LocalDateTime checkTime) {
        if (source.getLastReceivedAt() == null) {
            return "NOT SEEN";
        }

        Duration timeDifference = Duration.between(source.getLastReceivedAt(), checkTime);
        if (timeDifference.compareTo(Duration.ofSeconds(15)) <= 0) {
            return "ACTIVE";
        } else if (timeDifference.compareTo(Duration.ofSeconds(60)) <= 0) {
            return "STALE";
        } else {
            return "DISCONNECTED";
        }
    }

    // Check health of all registered sources
    public static void checkSourceHealth(String trackNumber) {
        List<Source> sources = SourceRepository.getAllSources();
        if (sources == null || sources.isEmpty()) {
            return;
        }
        LocalDateTime checkTime = LocalDateTime.now();
        for (Source source : sources) {
            String oldStatus = source.getSourceStatus();
            String newStatus = calculateSourceStatus(source, checkTime);
            if (!newStatus.equals(oldStatus)) {
                Alert alert = null;
                if (newStatus.equals("STALE")) {
                    alert = new Alert(trackNumber, "Source " + source.getSourceId() + " has become stale.", "MEDIUM", checkTime, source.getSourceId(), "Source becomes stale");
                } else if (newStatus.equals("DISCONNECTED")) {
                    alert = new Alert(trackNumber, "Source " + source.getSourceId() + " has become disconnected.", "HIGH", checkTime, source.getSourceId(), "Source becomes disconnected");
                }

                // Update status and create alert transactionally
                boolean updated = SourceRepository.updateSourceStatus(source.getSourceId(), oldStatus, newStatus, checkTime, alert);
                if (updated) {
                    source.setSourceStatus(newStatus);
                    System.out.println("Source " + source.getSourceId() + " changed status from " + oldStatus + " to " + newStatus);
                }
            }
        }
    }

    // Retrieve all alerts
    public static List<Source> getAllSources() {
        List <Source> listOfSources = SourceRepository.getAllSources();
        System.out.println("\nRetrieving all sources: ");
        if (listOfSources != null)
            for (Source source : listOfSources) {
                System.out.println(source);
            }
        else
            System.out.println("No available sources");
        return listOfSources;
    }
}

public abstract class BaseServlet extends HttpServlet {

    // Set UTF-8 request encoding before reading POST parameters
    protected void setRequestEncoding(HttpServletRequest request) throws IOException {
        request.setCharacterEncoding("UTF-8");
    }

    // Set UTF-8 response encoding
    protected void setResponseEncoding(HttpServletResponse response) {
        response.setCharacterEncoding("UTF-8");
    }

    // Handle bad input (HTTP 400)
    protected void sendBadRequest(HttpServletResponse response, String message) throws IOException {
        response.setStatus(HttpServletResponse.SC_BAD_REQUEST);
        response.setContentType("text/plain;charset=UTF-8");
        response.getWriter().println("Bad Request: " + message);
    }

    // Handle unexpected server errors (HTTP 500)
    protected void sendServerError(HttpServletResponse response, String message) throws IOException {
        response.setStatus(HttpServletResponse.SC_INTERNAL_SERVER_ERROR);
        response.setContentType("text/plain;charset=UTF-8");
        response.getWriter().println("Internal Server Error: " + message);
    }

    // Set the default response encoding for all requests
    @Override
    protected void service(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        setResponseEncoding(response);
        super.service(request, response);
    }
}

public class Server {
    public static void recevingDataFromClient()  {

        // creating socket to listen for data
        DatagramSocket datagramSocket = null;
        try {
            datagramSocket = new DatagramSocket(1234);
        } catch (SocketException e) {
            throw new RuntimeException(e);
        }

        // creating buffer
        byte[] receive = new byte[65535];

        // creating packet to receive data
        DatagramPacket datagramPacket = null;

        while (true) {
            datagramPacket = new DatagramPacket(receive, receive.length);

            //retrieving data
            try {
                datagramSocket.receive(datagramPacket);
            } catch (IOException e) {
                throw new RuntimeException(e);
            }

            if (byteToString(receive).toString().equals("end")){
                break;
            }

            TrackService.processTrackString(byteToString(receive));

            // clear buffer
            receive = new byte[65535];
        }
    }

    // convert
    private static String byteToString(byte[] data) {
        if (data == null)
            return null;
        StringBuilder string = new StringBuilder();
        int i = 0;
        while(data[i] != 0){
            string.append((char) data[i]);
            i++;
        }
        return string.toString();
    }
}

public class TrackService {
    static TrackRepository trackRepository = new TrackRepository();
    static TrackAnalyticsService trackAnalyticsService = new TrackAnalyticsService();
    private static List<Track> listOfTracks = null;
    private static List<Track> outOfOrderTracks = new ArrayList<Track>();
    // calculating duplicates and out-of-order messages
    private static int duplicatesCount = 0;
    private static int orderCount = 0;

    // Validate and process each track
    public static void processTrackString(String trackString) {

        // parsing track from string
        List<String> malformedErrors = ValidationService.parseTrackString(trackString);

        // getting source id and track number from malformed track
        if (!malformedErrors.isEmpty()) {
            String sourceId = null;
            String trackNumber = null;
            try{
                sourceId = trackString.substring(trackString.indexOf("LINK"), trackString.indexOf("LINK")+6);
            } catch (Exception e) {
                malformedErrors.add("Source Id not found");
            }
            try{
                trackNumber = trackString.substring(trackString.indexOf("TRK"), trackString.indexOf("TRK")+7);
            } catch (Exception e) {
                malformedErrors.add("Track Number not found");
            }
            // creating alert
            AlertService.processAlert(sourceId, trackNumber, String.join(", ", malformedErrors), "Malformed message");
        } else {

            // parsing and sending track object to processing
            Track track = parsingTrackString(trackString);
            processTrack(track);
        }
    }

    //overloading
    // Validate and process each track object
    public static boolean processTrack(Track track) {

        List<String> validationErrors = ValidationService.validateTrack(track);

        if (validationErrors.isEmpty()) {
            if (!hasDuplicates(track)) {
                if(!isOutOfOrder(track)) {
                    trackRepository.addTrack(track);
                    SourceService.processSources(track);
                    return true;
                }
            }
        } else {
            // creating alert
            AlertService.processAlert(track.getSourceId(), track.getTrackNumber(), String.join(", ", validationErrors), "Track validation failure");
        }
        return false;
    }

    // Checking for duplicate tracks
    public static boolean hasDuplicates(Track track){
        listOfTracks = findAllTracks();
        if (listOfTracks != null)
            for (Track oldTrack : listOfTracks){
                if ((oldTrack.getSourceId().equals(track.getSourceId())) &&
                        (oldTrack.getSequenceNumber() == track.getSequenceNumber())){
                    String alertMessage = "Message with the same Source Id and Sequence Number as an existing Track";
                    AlertService.processAlert(track.getSourceId(), track.getTrackNumber(), alertMessage, "Duplicate message");
                    duplicatesCount++;
                    return true;
                }
            }
        return false;
    }

    // storing out of order tracks
    public static boolean isOutOfOrder(Track track){
        Track latestTrack = trackRepository.getLatestTrackFromSource(track.getSourceId());
        if (latestTrack != null)
            if (latestTrack.getSequenceNumber() > track.getSequenceNumber()){
                String alertMessage = "Message with the same Source Id has a higher Sequence Number than an existing Track";
                AlertService.processAlert(track.getSourceId(), track.getTrackNumber(), alertMessage, "Out-of-order message");
                outOfOrderTracks.add(track);
                orderCount++;
                return true;
            }
        return false;
    }

    // getting list of out of order tracks
    public static List<Track> GetOutOfOrderTracks(){
        return outOfOrderTracks;
    }

    public static Track parsingTrackString (String trackString){
        // splitting string into segments
        String[] stringSeg = trackString.split("\\|");
        return new Track(
            stringSeg[2].trim(), //trackNumber
            Double.parseDouble(stringSeg[3].trim()), //latitude
            Double.parseDouble(stringSeg[4].trim()), //longitude
            Double.parseDouble(stringSeg[5].trim()), //speed
            stringSeg[6].trim(), //identity
            stringSeg[7].trim(), //trackStatus
            stringSeg[0].trim(), //sourceId
            Integer.parseInt(stringSeg[1].trim()), //sequenceNumber
            LocalDateTime.parse(stringSeg[8].trim()), //observationTime
            LocalDateTime.now()); //receivedTime
    }

    // Generate processing counter
    public static Map<String, Integer> processingCounter (){
        Map<String, Integer> processCounter = new LinkedHashMap<>();

        // calculating accepted unique messages
        int acceptedCount = 0;
        try{
            acceptedCount  = findAllTracks().size() + orderCount;
        } catch (Exception ignored) {}
        processCounter.put("Accepted Messages", acceptedCount);

        // calculating rejected messages
        int rejectedCount = 0;
        try{
            rejectedCount  = AlertService.getAllAlerts().size();
        } catch (Exception ignored) {

        }
        processCounter.put("Rejected Messages", rejectedCount);

        // calculating received messages
        processCounter.put("Received Messages", acceptedCount + rejectedCount + duplicatesCount);

        // add duplicate count
        processCounter.put("Duplicate Messages", duplicatesCount);

        // calculating processing failure
        processCounter.put("Processing Failure", duplicatesCount + rejectedCount);

        // Displaying counter
        if (processCounter != null) {
            System.out.println("\nDisplaying Processing Count: ");
            for (Map.Entry<String, Integer> entry : processCounter.entrySet()) {
                System.out.println(entry.getKey() + ": " + entry.getValue());
            }
        }

        return processCounter;
    }

    // Generating cross-source correlation tracks
    public static List<String> getCorrelatedTracks(){
        List<String> correlatedTracksInfo = new ArrayList<String>();
        listOfTracks = findAllTracks();
        // each track compared to other tracks
        for (Track track : listOfTracks) {
            for (Track track1 : listOfTracks) {
                if (track != track1) {
                    Duration timeDifference = Duration.between(track.getObservationTime(),
                            track1.getObservationTime());
                    double speedDifference = Math.abs(track.getSpeed() - track1.getSpeed());
                    double distanceDifference = calculateDistance(track.getLatitude(), track.getLongitude(),
                            track1.getLatitude(), track1.getLongitude());

                    // filtering tracks to find matches
                    // not yet done: fixing time and double results
                    if(
                            (track.getTrackNumber().equals(track1.getTrackNumber())) &&
                            (!track.getSourceId().equals(track1.getSourceId())) &&
                            (track.getIdentity().equals(track1.getIdentity())) &&
                            (speedDifference < 100) &&
                            (distanceDifference < 2) &&
                            ((timeDifference.compareTo(Duration.ofSeconds(30)) <= 0) && timeDifference.getSeconds()>=0)
                    ){
                        correlatedTracksInfo.add(
                                "\nTracks SourceId: " + track.getSourceId() + ", " + track1.getSourceId() +
                                "\nTrack Number: " + track.getTrackNumber() +
                                "\nFirst Track Position: " + track.getLatitude() + ", " + track.getLongitude() +
                                "\nSecond Track Position: " + track1.getLatitude() + ", " + track1.getLongitude() +
                                "\nTime Difference: " + timeDifference.getSeconds()+ " seconds" +
                                "\nSpeed Difference: " + speedDifference +
                                "\nDistance: " + distanceDifference + " km"
                        );
                    }
                }
            }
        }
        System.out.println("Displaying matching tracks");
        for(String matches : correlatedTracksInfo)
            System.out.println(matches);

        return correlatedTracksInfo;
    }

    // calculation to find the distance between two points
    public static double calculateDistance(double latitude1, double longtitude1, double latitude2, double longtitude2){
        double radLatitude1 = Math.toRadians(latitude1);
        double radLongtitude1 = Math.toRadians(longtitude1);
        double radLatitude2 = Math.toRadians(latitude2);
        double radLongtitude2 = Math.toRadians(longtitude2);
        double xAxis = (radLongtitude2 - radLongtitude1) * Math.cos((radLatitude1 + radLatitude2)/2);
        double yAxis = (radLatitude2 - radLatitude1);
        return Math.sqrt(xAxis*xAxis + yAxis*yAxis) * 6371; //earth radius
    }

public static List<String> parseTrackString(String trackString) {
        List<String> errors = new ArrayList<>();

        // splitting string into segments
        String[] stringSeg = trackString.split("\\|");

        // checking the use of correct partitioning
        if (!trackString.contains("|")) {
            errors.add("Message format is invalid");
        }


        // checking number of segments
        if (stringSeg.length != 9) {
            errors.add("Message missing fields");
        }

        try{
            Double.parseDouble(stringSeg[3].trim()); //latitude
        } catch (Exception e) {
            errors.add("Invalid latitude format");
        }

        try{
            Double.parseDouble(stringSeg[4].trim()); //longitude
        } catch (Exception e) {
            errors.add("Invalid longitude format");
        }

        try{
            Double.parseDouble(stringSeg[5].trim()); //speed
        } catch (Exception e) {
            errors.add("Invalid speed format");
        }

        try{
            LocalDateTime.parse(stringSeg[8].trim()); //observationTime
        } catch (Exception e) {
            errors.add("Invalid observation time format");
        }

        return errors;
    }

            default -> null;

    public static List<Track> advancedFilter(String sourceId, String trackNumber, String identity, String trackStatus, Double minSpeed, Double maxSpeed, LocalDateTime startTime, LocalDateTime endTime) {

        try {
// building the sql query using builder
            String sql = "SELECT * FROM tracks WHERE 1=1";

// storing parameter values in a list - to then run a loop on it
            List<Object> parameters = new ArrayList<>();

            if (sourceId != null && !sourceId.isEmpty()) {
                sql += " AND source_id = ?";
                parameters.add(sourceId);
            }

            if (trackNumber != null && !trackNumber.isEmpty()) {
                sql += " AND track_number = ?";
                parameters.add(trackNumber);
            }

            if (identity != null && !identity.isEmpty()) {
                sql += " AND identity_type = ?";
                parameters.add(identity);
            }

            if (trackStatus != null && !trackStatus.isEmpty()) {
                sql += " AND track_status = ?";
                parameters.add(trackStatus);
            }

            if (minSpeed != null) {
                sql += " AND speed >= ?";
                parameters.add(minSpeed);
            }

            if (maxSpeed != null) {
                sql += " AND speed <= ?";
                parameters.add(maxSpeed);
            }

            if (startTime != null) {
                sql += " AND observation_time >= ?";
                parameters.add(Timestamp.valueOf(startTime));
            }

            if (endTime != null) {
                sql += " AND observation_time <= ?";
                parameters.add(Timestamp.valueOf(endTime));
            }

//prep sql query
            PreparedStatement ps = connection.prepareStatement(sql);

// bind the values to the sql place holders ?
            for (int i = 0; i < parameters.size(); i++) {
                ps.setObject(i + 1, parameters.get(i));
            }

//execute the query
ResultSet resultSet = ps.executeQuery();

//store the matching tracks
//
List<Track> listOfTracks = new ArrayList<>();
            while (resultSet.next()) {
                listOfTracks.add(new Track(
                        resultSet.getInt(1),
                        resultSet.getString(2),
                        resultSet.getDouble(3),
                        resultSet.getDouble(4),
                        resultSet.getDouble(5),
                        resultSet.getString(6),
                        resultSet.getString(7),
                        resultSet.getString(8),
                        resultSet.getInt(9),
                        resultSet.getTimestamp(10).toLocalDateTime(),
                        resultSet.getTimestamp(11).toLocalDateTime()
                ));
            }

            resultSet.close();
            ps.close();

            return listOfTracks;

        } catch (SQLException e) {
            e.printStackTrace();
            throw new RuntimeException(
                    "Error executing advanced filter query", e); //error handling
        }
    }

package com.mtms.web.servlet;

import com.domain.com.mtms.Track;
import com.repository.com.mtms.TrackRepository;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;
import java.time.LocalDateTime;
import java.util.List;

public class AdvancedFiltersServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest request,
                         HttpServletResponse response)
            throws ServletException, IOException {

        request.setCharacterEncoding("UTF-8");
        response.setContentType("text/html;charset=UTF-8");

        PrintWriter out = response.getWriter();

        try {
            // get all filter values
            String sourceId = request.getParameter("sourceId");
            String trackNumber = request.getParameter("trackNumber");
            String identity = request.getParameter("identity");
            String trackStatus = request.getParameter("status");
            String minSpeedValue = request.getParameter("minSpeed");
            String maxSpeedValue = request.getParameter("maxSpeed");
            String startTimeValue = request.getParameter("startTime");
            String endTimeValue = request.getParameter("endTime");

            // converting empty values into null
            if (sourceId != null && sourceId.trim().isEmpty())
                sourceId = null;

            if (trackNumber != null && trackNumber.trim().isEmpty())
                trackNumber = null;

            if (identity != null && identity.trim().isEmpty())
                identity = null;

            if (trackStatus != null && trackStatus.trim().isEmpty())
                trackStatus = null;

            // converting speed into double
            Double minSpeed = null;
            Double maxSpeed = null;

            if (minSpeedValue != null && !minSpeedValue.trim().isEmpty())
                minSpeed = Double.parseDouble(minSpeedValue.trim());

            if (maxSpeedValue != null && !maxSpeedValue.trim().isEmpty())
                maxSpeed = Double.parseDouble(maxSpeedValue.trim());

            // converting date values into LocalDateTime
            LocalDateTime startTime = null;
            LocalDateTime endTime = null;

            if (startTimeValue != null && !startTimeValue.trim().isEmpty())
                startTime = LocalDateTime.parse(startTimeValue.trim());

            if (endTimeValue != null && !endTimeValue.trim().isEmpty())
                endTime = LocalDateTime.parse(endTimeValue.trim());

            // checking to ensure at least one filter is selected by user
            if (sourceId == null && trackNumber == null &&
                    identity == null && trackStatus == null &&
                    minSpeed == null && maxSpeed == null &&
                    startTime == null && endTime == null) {

                out.println("Please select at least one filter");
                return;
            }

            // validation to make sure minimum speed is not above maximum speed
            if (minSpeed != null && maxSpeed != null &&
                    minSpeed > maxSpeed) {

                out.println("Minimum speed is more than maximum speed");
                return;
            }

            // validation to make sure that start time is not after end time
            if (startTime != null && endTime != null &&
                    startTime.isAfter(endTime)) {

                out.println("Start time is after end time");
                return;
            }

// calling repo to check matching tracks

            List<Track> tracks = TrackRepository.advancedFilter(sourceId, trackNumber, identity, trackStatus, minSpeed, maxSpeed, startTime, endTime);

            //Displaying search results

            out.println("\n========== ADVANCED FILTER RESULTS ==========\n");

            if (tracks == null || tracks.isEmpty()) {
                out.println("No Matching Tracks Found");
            } else {
                for (Track track : tracks) {
                    out.println("---------------------------------------------");
                    out.println("Track Number      : " + track.getTrackNumber());
                    out.println("Source ID         : " + track.getSourceId());
                    out.println("Latitude          : " + track.getLatitude());
                    out.println("Longitude         : " + track.getLongitude());
                    out.println("Speed             : " + track.getSpeed());
                    out.println("Identity          : " + track.getIdentity());
                    out.println("Track Status      : " + track.getTrackStatus());
                    out.println("Sequence Number   : " + track.getSequenceNumber());
                    out.println("Observation Time  : " + track.getObservationTime());
                }

                out.println("---------------------------------------------");
            }

            out.println("=============================================\n");

        } catch (NumberFormatException e) {

            out.println("Please enter valid speed values");

        } catch (java.time.format.DateTimeParseException e) {

            out.println("Please enter valid date and time values");

        } catch (Exception e) {

            System.out.println("Cannot apply advanced track search: " + e);
            e.printStackTrace();

            out.println("Cannot apply track search");
        }
    }
}

  public static List<String> validateTrack(Track track) {
        List<String> errors = new ArrayList<>();

        // Validation to Check Track number
        if (track.getTrackNumber() == null ||
                track.getTrackNumber().trim().isEmpty()) {
            errors.add("Track Number is Empty");
        } else if (track.getTrackNumber().trim().length() < 6)
        {
            errors.add("Track Number is Too Short");
        }

        // Validation to Check Source Id
        if (track.getSourceId() == null ||
                track.getSourceId().trim().isEmpty()) {
            errors.add("Source Id is Empty");
        } else if (track.getSourceId().trim().length() < 4)
        {
            errors.add("Source Id is Too Short");
        }

        // Validation to Check speed (>0 and <=2000)
        if (track.getSequenceNumber() <= 0) {
            errors.add("Sequence Number is Invalid ");
        }

        // Validation to Check Latitude (-90 to 90)
        if (track.getLatitude() < -90 ||
                track.getLatitude() > 90)
        {
            errors.add("Latitude is Invalid ");
        }

        // Validation to Check Longitude (-180 to 180)
        if (track.getLongitude() < -180 ||
                track.getLongitude() > 180)
        {
            errors.add("longitude is Invalid");
        }

        // Validation to Check speed (>0 and <=2000)
        if (track.getSpeed() < 0) {
            errors.add("Speed is Invalid ");
        } else if (track.getSpeed() > 2000)
        {
            errors.add("Speed is too high ");
